1. PROJECT OVERVIEW
You are building a Flutter mobile application called the Appointment App – Smart Slot Booking System. This is a full-stack, production-grade Flutter app developed progressively across 13 labs. The app allows users to register, log in, browse service providers, book appointment slots, manage their bookings, receive notifications, and view booking analytics — all powered by Firebase as the backend.

Core Identity:

App Name: Appointment App
Tagline: "Book smarter, wait less."
Platform: Flutter (Dart) — cross-platform Android & iOS
Backend: Firebase (Authentication + Firestore)
Primary Color: #1F4E79 (deep navy blue)
Secondary Color: #2E75B6
Accent Color: Used for selected state highlights
Font: Poppins (headings), system default (body)
Border Radius: 12px (TextFields), 14px (Cards), 20px (status badges), 10px (Snackbars)
Package Name (Android): com.d24it166.appointmentapp

7. DATA MODEL — APPOINTMENT (lib/models/appointment_model.dart)
class AppointmentModel {
  final String appointmentId;
  final String userId;
  final String serviceName;
  final String providerName;
  final String date;         // Format: yyyy-MM-dd
  final String timeSlot;     // Format: 10:00 AM
  final String status;       // pending | confirmed | cancelled
  final String notes;
  final String createdAt;    // ISO 8601

  AppointmentModel({
    required this.appointmentId,
    required this.userId,
    required this.serviceName,
    required this.providerName,
    required this.date,
    required this.timeSlot,
    this.status = 'pending',
    this.notes = '',
    required this.createdAt,
  });

  factory AppointmentModel.fromMap(Map<String, dynamic> map, String id) {
    return AppointmentModel(
      appointmentId: id,
      userId:       map['userId']       ?? '',
      serviceName:  map['serviceName']  ?? '',
      providerName: map['providerName'] ?? '',
      date:         map['date']         ?? '',
      timeSlot:     map['timeSlot']     ?? '',
      status:       map['status']       ?? 'pending',
      notes:        map['notes']        ?? '',
      createdAt:    map['createdAt']    ?? '',
    );
  }

  Map<String, dynamic> toMap() => {
    'userId':       userId,
    'serviceName':  serviceName,
    'providerName': providerName,
    'date':         date,
    'timeSlot':     timeSlot,
    'status':       status,
    'notes':        notes,
    'createdAt':    createdAt,
  };
}

AppointmentService (lib/services/appointment_service.dart)
All Firestore operations — separated from UI.

class AppointmentService {
  final _col = FirebaseFirestore.instance.collection('appointments');

  Future<void> createAppointment(AppointmentModel appt) async =>
      await _col.add(appt.toMap());

  Stream<List<AppointmentModel>> getAppointments(String userId) =>
      _col.where('userId', isEqualTo: userId)
          .orderBy('createdAt', descending: true)
          .snapshots()
          .map((snap) => snap.docs
              .map((d) => AppointmentModel.fromMap(d.data(), d.id))
              .toList());

  Future<void> updateAppointment(String id, Map<String, dynamic> data) async =>
      await _col.doc(id).update(data);

  Future<void> deleteAppointment(String id) async =>
      await _col.doc(id).delete();
}

Firestore Security Rules (recommended)
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    match /appointments/{appointmentId} {
      allow read, write: if request.auth != null
        && request.auth.uid == resource.data.userId;
      allow create: if request.auth != null
        && request.resource.data.userId == request.auth.uid;
    }
  }
}
